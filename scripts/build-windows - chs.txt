@echo off
setlocal EnableExtensions EnableDelayedExpansion
chcp 936 >nul

set "SCRIPT_DIR=%~dp0"
for %%I in ("%SCRIPT_DIR%..") do set "PROJECT_ROOT=%%~fI"
cd /d "%PROJECT_ROOT%"

echo.
echo ==========================================
echo DockerCopilot Windows 一键构建脚本
echo 项目目录: %PROJECT_ROOT%
echo ==========================================
echo.

call :read_version
call :resolve_build_date
call :ensure_node || goto :fail
call :ensure_go || goto :fail
call :choose_target || goto :fail
call :install_frontend_dependencies || goto :fail
call :build_frontend || goto :fail
call :build_binary || goto :fail
call :print_summary
goto :success

:read_version
set "BUILD_VERSION="
set /p BUILD_VERSION=<version
if not defined BUILD_VERSION set "BUILD_VERSION=dev"
goto :eof

:resolve_build_date
set "BUILD_DATE="
for /f "usebackq delims=" %%I in (`powershell -NoProfile -Command "(Get-Date).ToUniversalTime().ToString('yyyy-MM-ddTHH:mm:ssZ')"`) do set "BUILD_DATE=%%I"
if not defined BUILD_DATE set "BUILD_DATE=%DATE% %TIME%"
goto :eof

:ensure_node
echo [信息] 正在检查 Node.js / npm ...
where node >nul 2>nul && where npm.cmd >nul 2>nul && (
  for /f "delims=" %%I in ('node -v') do set "NODE_VERSION=%%I"
  echo [成功] 已找到 Node.js !NODE_VERSION!
  goto :eof
)

for /d %%D in ("%LOCALAPPDATA%\Microsoft\WinGet\Packages\OpenJS.NodeJS.LTS_Microsoft.Winget.Source_8wekyb3d8bbwe\node-v*-win-x64") do (
  set "PATH=%%~fD;!PATH!"
)

where node >nul 2>nul && where npm.cmd >nul 2>nul && (
  for /f "delims=" %%I in ('node -v') do set "NODE_VERSION=%%I"
  echo [成功] 已自动补全 Node.js 路径: !NODE_VERSION!
  goto :eof
)

echo [错误] 未找到 Node.js 或 npm.cmd，请先安装 Node.js 20+！
exit /b 1

:ensure_go
echo [信息] 正在检查 Go ...
where go >nul 2>nul && (
  for /f "delims=" %%I in ('go version') do set "GO_VERSION=%%I"
  echo [成功] 已找到 !GO_VERSION!
  goto :eof
)

if exist "%ProgramFiles%\Go\bin\go.exe" set "PATH=%ProgramFiles%\Go\bin;!PATH!"
if exist "%LocalAppData%\Programs\Go\bin\go.exe" set "PATH=%LocalAppData%\Programs\Go\bin;!PATH!"

where go >nul 2>nul && (
  for /f "delims=" %%I in ('go version') do set "GO_VERSION=%%I"
  echo [成功] 已自动补全 Go 路径: !GO_VERSION!
  goto :eof
)

echo [错误] 未找到 go.exe，请先安装 Go 1.24.1+！
exit /b 1

:choose_target
echo.
echo 请选择要构建的目标：
echo   1^) Windows amd64 EXE
echo   2^) Linux amd64
echo   3^) Linux arm64
echo.
set /p BUILD_CHOICE=请输入选项 [1/2/3]: 

if "%BUILD_CHOICE%"=="1" (
  set "TARGET_GOOS=windows"
  set "TARGET_GOARCH=amd64"
  set "TARGET_NAME=dockerCopilot.exe"
  set "TARGET_DIR=%PROJECT_ROOT%\release\windows\amd64"
  goto :eof
)
if "%BUILD_CHOICE%"=="2" (
  set "TARGET_GOOS=linux"
  set "TARGET_GOARCH=amd64"
  set "TARGET_NAME=dockerCopilot"
  set "TARGET_DIR=%PROJECT_ROOT%\release\linux\amd64"
  goto :eof
)
if "%BUILD_CHOICE%"=="3" (
  set "TARGET_GOOS=linux"
  set "TARGET_GOARCH=arm64"
  set "TARGET_NAME=dockerCopilot"
  set "TARGET_DIR=%PROJECT_ROOT%\release\linux\arm64"
  goto :eof
)

echo [错误] 选项无效，请重新运行脚本并输入 1、2 或 3！
exit /b 1

:install_frontend_dependencies
echo.
echo [信息] 正在安装 PC 前端依赖...
call npm.cmd ci
if errorlevel 1 exit /b 1

echo [信息] 正在安装移动端前端依赖...
call npm.cmd --prefix .\web-mobile ci
if errorlevel 1 exit /b 1

echo [成功] 前端依赖安装完成
goto :eof

:build_frontend
echo.
echo [信息] 正在构建前端（PC + Mobile）...
call npm.cmd run build:front
if errorlevel 1 exit /b 1

echo [成功] 前端构建完成
goto :eof

:build_binary
echo.
echo [信息] 正在构建 %TARGET_GOOS%/%TARGET_GOARCH% 二进制...
if not exist "%TARGET_DIR%" mkdir "%TARGET_DIR%"

set "LDFLAGS=-w -s -X github.com/onlyLTY/dockerCopilot/internal/config.Version=%BUILD_VERSION% -X github.com/onlyLTY/dockerCopilot/internal/config.BuildDate=%BUILD_DATE%"
set "CGO_ENABLED=0"
set "GOOS=%TARGET_GOOS%"
set "GOARCH=%TARGET_GOARCH%"

go build -a --trimpath -ldflags "!LDFLAGS!" -o "%TARGET_DIR%\%TARGET_NAME%" .
if errorlevel 1 exit /b 1

echo [成功] 二进制构建完成: %TARGET_DIR%\%TARGET_NAME%
goto :eof

:print_summary
echo.
echo ==========================================
echo 构建完成
echo ------------------------------------------
echo 版本: %BUILD_VERSION%
echo 构建时间(UTC): %BUILD_DATE%
echo PC 前端目录: %PROJECT_ROOT%\front\pc
echo Mobile 前端目录: %PROJECT_ROOT%\front\mobile
echo 二进制文件: %TARGET_DIR%\%TARGET_NAME%
echo ==========================================
goto :eof

:fail
echo.
echo [错误] 构建失败
exit /b 1

:success
echo.
echo [成功] 全部任务执行完成
exit /b 0
